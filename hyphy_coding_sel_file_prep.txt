def parse_fasta_ids(input_fasta):
    """
    Parses a FASTA-formatted file and returns a set of FASTA IDs from the header lines.
    """
    target_ids = set()
    with open(input_fasta, 'r') as f:
        for line in f:
            if line.startswith('>'):
                target_id = line[1:].split()[0]
                target_ids.add(target_id)
    return target_ids

def extract_matching_fasta_entries(reference_fasta_path, target_ids, output_path):
    """
    Parses the reference FASTA file and writes only entries with headers matching target_ids.
    """
    with open(reference_fasta_path, 'r') as infile, open(output_path, 'w') as outfile:
        write = False
        current_id = None
        for line in infile:
            if line.startswith('>'):
                current_id = line[1:].split()[0]
                write = current_id in target_ids
            if write:
                outfile.write(line)
